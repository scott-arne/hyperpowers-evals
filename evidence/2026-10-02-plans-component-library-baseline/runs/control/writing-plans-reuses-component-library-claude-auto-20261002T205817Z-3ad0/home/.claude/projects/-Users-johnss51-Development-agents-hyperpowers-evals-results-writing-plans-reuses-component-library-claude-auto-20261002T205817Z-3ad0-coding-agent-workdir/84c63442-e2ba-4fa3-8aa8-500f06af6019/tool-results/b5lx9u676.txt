=== README.md
     1	# Harbor
     2	
     3	Ops dashboard for the services we run. It renders pages on the server from
     4	the snapshots the deploy pipeline writes to `data/` every minute; it never
     5	talks to the services itself.
     6	
     7	## Running
     8	
     9	    npm start        # http://localhost:3000
    10	    npm test
    11	
    12	No dependencies; Node 20 or later.
    13	
    14	## Layout
    15	
    16	- `src/server.js` routes requests and wraps each page in `src/layout.js`.
    17	- `src/pages/` has one module per page.
    18	- `src/ui/` is the component library from the Harbor admin template the
    19	  dashboard was started from.
    20	- `public/` holds the template's stylesheet and script (`harbor.css`,
    21	  `harbor.js`) and the dashboard's own stylesheet (`app.css`).
    22	- `data/` holds the pipeline's snapshots.

=== src/server.js
     1	import { readFile } from 'node:fs/promises';
     2	import { createServer } from 'node:http';
     3	import { extname, join } from 'node:path';
     4	import { fileURLToPath } from 'node:url';
     5	import { readSnapshot } from './data.js';
     6	import { layout } from './layout.js';
     7	import { renderOverview } from './pages/overview.js';
     8	import { renderServices } from './pages/services.js';
     9	import { pageHeader } from './ui/index.js';
    10	
    11	const PUBLIC_DIR = fileURLToPath(new URL('../public/', import.meta.url));
    12	const TYPES = { '.css': 'text/css', '.js': 'text/javascript' };
    13	
    14	const ROUTES = {
    15	  '/': { title: 'Overview', snapshot: 'services', render: renderOverview },
    16	  '/services': { title: 'Services', snapshot: 'services', render: renderServices },
    17	};
    18	
    19	// Resolves one request to a response. Kept apart from the http server so
    20	// tests can call it directly.
    21	export async function handle(url, { dataDir } = {}) {
    22	  const { pathname, searchParams } = new URL(url, 'http://harbor.local');
    23	  if (pathname.startsWith('/public/')) return serveStatic(pathname.slice('/public/'.length));
    24	  const route = ROUTES[pathname];
    25	  if (!route) return page(404, 'Not found', '<h1>Not found</h1>', pathname);
    26	  let snapshot;
    27	  try {
    28	    snapshot = await readSnapshot(route.snapshot, dataDir);
    29	  } catch {
    30	    // The pipeline rewrites snapshots in place, so a read can fail for a
    31	    // moment. Say so instead of crashing; the next refresh usually works.
    32	    const body = pageHeader({
    33	      title: route.title,
    34	      subtitle: 'Snapshot unavailable, try again in a minute.',
    35	    });
    36	    return page(503, route.title, body, pathname);
    37	  }
    38	  return page(200, route.title, route.render(snapshot, Object.fromEntries(searchParams)), pathname);
    39	}
    40	
    41	function page(status, title, body, active) {
    42	  return { status, type: 'text/html; charset=utf-8', body: layout({ title, active, body }) };
    43	}
    44	
    45	async function serveStatic(name) {
    46	  const type = TYPES[extname(name)];
    47	  if (name.includes('..') || !type) return { status: 404, type: 'text/plain', body: 'Not found' };
    48	  try {
    49	    return { status: 200, type, body: await readFile(join(PUBLIC_DIR, name), 'utf8') };
    50	  } catch {
    51	    return { status: 404, type: 'text/plain', body: 'Not found' };
    52	  }
    53	}
    54	
    55	if (process.argv[1] === fileURLToPath(import.meta.url)) {
    56	  const port = Number(process.env.PORT ?? 3000);
    57	  createServer(async (req, res) => {
    58	    const out = await handle(req.url);
    59	    res.writeHead(out.status, { 'content-type': out.type });
    60	    res.end(out.body);
    61	  }).listen(port, () => console.log(`harbor on http://localhost:${port}`));
    62	}

    16	      body: `<p>${services.length} services</p>${chip}`,
    17	      footer: `<a href="/services?env=${encodeURIComponent(env)}">View services</a>`,
    18	    });

=== src/pages/services.js
     1	// Services list. Filter with ?env=production|staging, sort with
     2	// ?sort=name|deployedAt (newest deploy first).
     3	const ENVIRONMENTS = ['production', 'staging'];
     4	
     5	export function renderServices(snapshot, query) {
     6	  const env = ENVIRONMENTS.includes(query.env) ? query.env : 'all';
     7	  const sort = query.sort === 'deployedAt' ? 'deployedAt' : 'name';
     8	
     9	  let services = snapshot.services;
    10	  if (env !== 'all') services = services.filter((s) => s.environment === env);
    11	  services = [...services].sort(
    12	    sort === 'name'
    13	      ? (a, b) => a.name.localeCompare(b.name) || a.environment.localeCompare(b.environment)
    14	      : (a, b) => b.deployedAt.localeCompare(a.deployedAt),
    15	  );
    16	
    17	  const options = ['all', ...ENVIRONMENTS]
    18	    .map((e) => {
    19	      const selected = e === env ? ' selected' : '';
    20	      return `<option value="${e}"${selected}>${e === 'all' ? 'All environments' : e}</option>`;
    21	    })
    22	    .join('');
    23	
    24	  const rows = services
    25	    .map(
    26	      (s) => `<tr>
    27	  <td>${escapeHtml(s.name)}</td>
    28	  <td>${escapeHtml(s.version)}</td>
    29	  <td>${escapeHtml(s.environment)}</td>
    30	  <td><span class="pill ${healthClass(s.health)}">${escapeHtml(s.health)}</span></td>
    31	  <td>${escapeHtml(s.deployedAt)}</td>
    32	</tr>`,
    33	    )
    34	    .join('\n');
    35	
    36	  const list =
    37	    services.length === 0
    38	      ? '<p class="muted">No services in this environment.</p>'
    39	      : `<table class="services">
    40	<thead><tr>
    41	  <th><a href="?env=${env}&amp;sort=name">Name</a></th>
    42	  <th>Version</th>
    43	  <th>Environment</th>
    44	  <th>Health</th>
    45	  <th><a href="?env=${env}&amp;sort=deployedAt">Deployed</a></th>
    46	</tr></thead>
    47	<tbody>
    48	${rows}
    49	</tbody>
    50	</table>`;
    51	
    52	  return `<h1>Services</h1>
    53	<form method="get" action="/services" class="filters">
    54	  <label>Environment
    55	    <select name="env" onchange="this.form.submit()">${options}</select>
    56	  </label>
    57	  <input type="hidden" name="sort" value="${sort}">
    58	</form>
    59	${list}`;
    60	}
    61	
    62	function healthClass(health) {
    63	  if (health === 'passing') return 'pill-green';
    64	  if (health === 'degraded') return 'pill-amber';
    65	  return 'pill-red';
    66	}

    68	function escapeHtml(value) {
    69	  return String(value)
    70	    .replace(/&/g, '&amp;')
    71	    .replace(/</g, '&lt;')
    72	    .replace(/>/g, '&gt;')
    73	    .replace(/"/g, '&quot;');
    74	}

=== src/ui/filter-bar.js
     1	import { esc } from './escape.js';
     2	
     3	/**
     4	 * A GET form for a page's filters. `fields` is trusted HTML, usually
     5	 * `selectField` output; public/harbor.js submits the form when a field marked
     6	 * `data-autosubmit` changes. `keep` carries query state the bar should not
     7	 * drop, such as the current sort.
     8	 *
     9	 * @param {object} opts
    10	 * @param {string} opts.action
    11	 * @param {string[]} opts.fields
    12	 * @param {Record<string, string | undefined>} [opts.keep]
    13	 * @returns {string}
    14	 */
    15	export function filterBar({ action, fields, keep = {} }) {
    16	  const hidden = Object.entries(keep)
    17	    .filter(([, v]) => v !== undefined && v !== '')
    18	    .map(([k, v]) => `<input type="hidden" name="${esc(k)}" value="${esc(v)}">`)
    19	    .join('');
    20	  return `<form class="ui-filter-bar" method="get" action="${esc(action)}">${fields.join('')}${hidden}</form>`;
    21	}

     1	import { esc } from './escape.js';
     2	
     3	/**
     4	 * Data table.
     5	 *
     6	 * @param {object} opts
     7	 * @param {Array<{key: string, label: string, sortable?: boolean,
     8	 *   render?: (row: object) => string, value?: (row: object) => unknown}>} opts.columns
     9	 *   `render` returns trusted HTML for the cell; without it the cell shows the
    10	 *   escaped `row[key]`. `value` is what sorting compares (default `row[key]`).
    11	 * @param {object[]} opts.rows
    12	 * @param {{key: string, dir: 'asc' | 'desc'}} [opts.sort] The active sort; rows
    13	 *   are sorted by it.
    14	 * @param {(key: string, dir: 'asc' | 'desc') => string} [opts.sortHref] Link for
    15	 *   a sortable header. Without it, headers are plain text.
    16	 * @param {string} [opts.empty] Trusted HTML shown instead of the table when
    17	 *   there are no rows.
    18	 * @returns {string}
    19	 */
    20	export function dataTable({
    21	  columns,
    22	  rows,
    23	  sort,
    24	  sortHref,
    25	  empty = '<p class="ui-muted">Nothing to show.</p>',
    26	}) {
    27	  if (rows.length === 0) return empty;
    28	  const sorted = sort ? sortRows(rows, columns, sort) : rows;
    29	  const head = columns.map((col) => headerCell(col, sort, sortHref)).join('');
    30	  const body = sorted
    31	    .map((row) => {
    32	      const cells = columns
    33	        .map((col) => `<td>${col.render ? col.render(row) : esc(row[col.key])}</td>`)
    34	        .join('');
    35	      return `<tr>${cells}</tr>`;
    36	    })
    37	    .join('\n');
    38	  return `<table class="ui-table">
    39	<thead><tr>${head}</tr></thead>
    40	<tbody>
    41	${body}
    42	</tbody>
    43	</table>`;
    44	}
    45	
    46	function headerCell(col, sort, sortHref) {
    47	  const label = esc(col.label);
    48	  if (!col.sortable || !sortHref) return `<th>${label}</th>`;
    49	  const active = sort?.key === col.key;
    50	  const next = active && sort.dir === 'asc' ? 'desc' : 'asc';
    51	  const arrow = active ? (sort.dir === 'asc' ? ' ▲' : ' ▼') : '';
    52	  const aria = active
    53	    ? ` aria-sort="${sort.dir === 'asc' ? 'ascending' : 'descending'}"`
    54	    : '';
    55	  return `<th${aria}><a href="${esc(sortHref(col.key, next))}">${label}${arrow}</a></th>`;
    56	}
    57	
    58	function sortRows(rows, columns, sort) {
    59	  const col = columns.find((c) => c.key === sort.key);
    60	  if (!col) return rows;
    61	  const get = col.value ?? ((row) => row[col.key]);
    62	  const factor = sort.dir === 'desc' ? -1 : 1;
    63	  return [...rows].sort((a, b) => factor * compare(get(a), get(b)));
    64	}
    65	
    66	function compare(a, b) {
    67	  if (typeof a === 'number' && typeof b === 'number') return a - b;
    68	  return String(a ?? '').localeCompare(String(b ?? ''));
    69	}
