=== README.md
     1	# Harbor
     2	
     3	Ops dashboard for the services we run. It renders pages on the server from
     4	the snapshots the deploy pipeline writes to `data/` every minute; it never
     5	talks to the services itself.
     6	
     7	## Running
     8	
     9	    npm start        # http://localhost:3000
    10	    npm test
    11	
    12	No dependencies; Node 20 or later.
    13	
    14	## Pages
    15	
    16	Overview, Services, Incidents, On-call and Runbooks, one module each in
    17	`src/pages/`. `src/server.js` routes requests and wraps each page in
    18	`src/layout.js`; `public/app.css` holds the dashboard's styles.

     1	import { readFile } from 'node:fs/promises';
     2	import { createServer } from 'node:http';
     3	import { extname, join } from 'node:path';
     4	import { fileURLToPath } from 'node:url';
     5	import { readSnapshot } from './data.js';
     6	import { escapeHtml } from './html.js';
     7	import { layout } from './layout.js';
     8	import { renderIncidents } from './pages/incidents.js';
     9	import { renderOncall } from './pages/oncall.js';
    10	import { renderOverview } from './pages/overview.js';
    11	import { renderRunbooks } from './pages/runbooks.js';
    12	import { renderServices } from './pages/services.js';
    13	
    14	const PUBLIC_DIR = fileURLToPath(new URL('../public/', import.meta.url));
    15	const TYPES = { '.css': 'text/css', '.js': 'text/javascript' };
    16	
    17	const ROUTES = {
    18	  '/': { title: 'Overview', snapshot: 'services', render: renderOverview },
    19	  '/services': { title: 'Services', snapshot: 'services', render: renderServices },
    20	  '/incidents': { title: 'Incidents', snapshot: 'incidents', render: renderIncidents },
    21	  '/oncall': { title: 'On-call', snapshot: 'oncall', render: renderOncall },
    22	  '/runbooks': { title: 'Runbooks', snapshot: 'runbooks', render: renderRunbooks },
    23	};
    24	
    25	// Resolves one request to a response. Kept apart from the http server so
    26	// tests can call it directly.
    27	export async function handle(url, { dataDir } = {}) {
    28	  const { pathname, searchParams } = new URL(url, 'http://harbor.local');
    29	  if (pathname.startsWith('/public/')) return serveStatic(pathname.slice('/public/'.length));
    30	  const route = ROUTES[pathname];
    31	  if (!route) return page(404, 'Not found', '<h1>Not found</h1>', pathname);
    32	  let snapshot;
    33	  try {
    34	    snapshot = await readSnapshot(route.snapshot, dataDir);
    35	  } catch {
    36	    // The pipeline rewrites snapshots in place, so a read can fail for a
    37	    // moment. Say so instead of crashing; the next refresh usually works.
    38	    const body = `<h1>${escapeHtml(route.title)}</h1>
    39	<p class="muted">Snapshot unavailable, try again in a minute.</p>`;
    40	    return page(503, route.title, body, pathname);
    41	  }
    42	  return page(200, route.title, route.render(snapshot, Object.fromEntries(searchParams)), pathname);
    43	}
    44	
    45	function page(status, title, body, active) {
    46	  return { status, type: 'text/html; charset=utf-8', body: layout({ title, active, body }) };
    47	}
    48	
    49	async function serveStatic(name) {
    50	  const type = TYPES[extname(name)];
    51	  if (name.includes('..') || !type) return { status: 404, type: 'text/plain', body: 'Not found' };
    52	  try {
    53	    return { status: 200, type, body: await readFile(join(PUBLIC_DIR, name), 'utf8') };
    54	  } catch {
    55	    return { status: 404, type: 'text/plain', body: 'Not found' };
    56	  }
    57	}
    58	
    59	if (process.argv[1] === fileURLToPath(import.meta.url)) {
    60	  const port = Number(process.env.PORT ?? 3000);
    61	  createServer(async (req, res) => {
    62	    const out = await handle(req.url);
    63	    res.writeHead(out.status, { 'content-type': out.type });
    64	    res.end(out.body);
    65	  }).listen(port, () => console.log(`harbor on http://localhost:${port}`));
    66	}

     4	// Incidents, newest first. Open ones by default; ?state=resolved shows the

    11

    13	    const current = value === state ? ' aria-current="page"' : '';
    14	    return `<a href="/incidents?state=${value}"${current}>${label}</a>`;
    15	  };
    16	
    17	  const items = incidents
    18	    .map((i) => {
    19	      const runbook = button({
    20	        label: 'Runbook',
    21	        href: `/runbooks#${encodeURIComponent(i.runbook)}`,
    22	        variant: 'link',

     1	import { button } from '#kit/button';
     2	import { dialog } from '#kit/dialog';
     3	import { escapeHtml } from '../html.js';
     4	
     5	// Services list. Filter with ?env=production|staging; sort with
     6	// ?sort=name|deployedAt and ?dir=asc|desc (default: name, ascending).
     7	const ENVIRONMENTS = ['production', 'staging'];
     8	const SORTS = {
     9	  name: (a, b) => a.name.localeCompare(b.name) || a.environment.localeCompare(b.environment),
    10	  deployedAt: (a, b) => a.deployedAt.localeCompare(b.deployedAt),
    11	};
    12	
    13	export function renderServices(snapshot, query) {
    14	  const env = ENVIRONMENTS.includes(query.env) ? query.env : 'all';
    15	  const sort = Object.hasOwn(SORTS, query.sort ?? '') ? query.sort : 'name';
    16	  const dir = query.dir === 'desc' ? 'desc' : 'asc';
    17	
    18	  let services = snapshot.services;
    19	  if (env !== 'all') services = services.filter((s) => s.environment === env);
    20	  const factor = dir === 'desc' ? -1 : 1;
    21	  services = [...services].sort((a, b) => factor * SORTS[sort](a, b));
    22	
    23	  const options = ['all', ...ENVIRONMENTS]
    24	    .map((e) => {
    25	      const selected = e === env ? ' selected' : '';
    26	      return `<option value="${e}"${selected}>${e === 'all' ? 'All environments' : e}</option>`;
    27	    })
    28	    .join('');
    29	
    30	  // Clicking the active column flips its direction; any other column starts
    31	  // ascending. The links carry the filter so sorting keeps it.
    32	  const sortHeader = (key, label) => {
    33	    const active = key === sort;
    34	    const next = active && dir === 'asc' ? 'desc' : 'asc';
    35	    const arrow = active ? (dir === 'asc' ? ' ▲' : ' ▼') : '';
    36	    const ariaSort = active ? ` aria-sort="${dir === 'asc' ? 'ascending' : 'descending'}"` : '';
    37	    return `<th${ariaSort}><a href="?env=${env}&amp;sort=${key}&amp;dir=${next}">${label}${arrow}</a></th>`;
    38	  };
    39	
    40	  const rows = services
    41	    .map(
    42	      (s) => `<tr>
    43	  <td>${escapeHtml(s.name)}</td>
    44	  <td>${escapeHtml(s.version)}</td>
    45	  <td>${escapeHtml(s.environment)}</td>
    46	  <td><span class="pill ${healthClass(s.health)}">${escapeHtml(s.health)}</span></td>
    47	  <td>${escapeHtml(s.deployedAt)}</td>
    48	</tr>`,
    49	    )
    50	    .join('\n');
    51	
    52	  const list =
    53	    services.length === 0
    54	      ? '<p class="muted">No services in this environment.</p>'
    55	      : `<table class="services">
    56	<thead><tr>
    57	  ${sortHeader('name', 'Name')}
    58	  <th>Version</th>
    59	  <th>Environment</th>
    60	  <th>Health</th>
    61	  ${sortHeader('deployedAt', 'Deployed')}
    62	</tr></thead>
    63	<tbody>
    64	${rows}
    65	</tbody>
    66	</table>`;
    67	
    68	  const legend = dialog({
    69	    id: 'health-legend',
    70	    title: 'Health checks',
    71	    body: `<ul class="legend">
    72	  <li><span class="pill pill-green">passing</span> every check passed in the last minute</li>
    73	  <li><span class="pill pill-amber">degraded</span> some checks failed</li>
    74	  <li><span class="pill pill-red">failing</span> most or all checks failed</li>
    75	</ul>`,
    76	  });
    77	  const legendButton = button({
    78	    label: 'Health legend',
    79	    variant: 'ghost',
    80	    size: 'sm',
    81	    attrs: { 'data-dialog-open': 'health-legend' },
    82	  });
    83	
    84	  return `<div class="title-row">
    85	  <h1>Services</h1>
    86	  ${legendButton}
    87	</div>
    88	<form method="get" action="/services" class="filters">
    89	  <label>Environment
    90	    <select name="env" onchange="this.form.submit()">${options}</select>
    91	  </label>
    92	  <input type="hidden" name="sort" value="${sort}">
    93	  <input type="hidden" name="dir" value="${dir}">
    94	</form>
    95	${list}
    96	${legend}`;
    97	}
    98	
    99	function healthClass(health) {
   100	  if (health === 'passing') return 'pill-green';
   101	  if (health === 'degraded') return 'pill-amber';
   102	  return 'pill-red';
   103	}
